QUESTION # 01: CREATE A CAR CLASS

class Car:

    def __init__(self, brand, color):
        self.brand = brand
        self.color = color

    def start(self):
        print("The car is starting.")


# Creating an object
car1 = Car("Toyota", "Black")

# Calling the method
car1.start()
 
OUTPUT: The car is starting

QUESTION NO 2: Create a Book Class

class Book:

    def __init__(self, title, author):
        self.title = title
        self.author = author

    def display_info(self):
        print("Book Title:", self.title)
        print("Author:", self.author)


# Creating three objects
book1 = Book("The Alchemist", "Paulo Coelho")
book2 = Book("The Kite Runner", "Khaled Hosseini")
book3 = Book("Harry Potter", "J.K. Rowling")

# Calling methods
book1.display_info()
print("------------------")

book2.display_info()
print("------------------")

book3.display_info()

OUTPUT:
Book Title: The Alchemist
Author: Paulo Coelho
------------------
Book Title: The Kite Runner
Author: Khaled Hosseini
------------------
Book Title: Harry Potter
Author: J.K. Rowling

QUESTION NO 03: Create a Calculator Class

class Calculator:

    def __init__(self, num1, num2):
        self.num1 = num1
        self.num2 = num2

    def addition(self):
        print("Addition:", self.num1 + self.num2)

    def subtraction(self):
        print("Subtraction:", self.num1 - self.num2)

    def multiplication(self):
        print("Multiplication:", self.num1 * self.num2)

    def division(self):
        if self.num2 != 0:
            print("Division:", self.num1 / self.num2)
        else:
            print("Cannot divide by zero!")


# Creating an object
calc = Calculator(20, 10)

# Calling all methods
calc.addition()
calc.subtraction()
calc.multiplication()
calc.division()

OUTPUT:
Addition: 30
Subtraction: 10
Multiplication: 200
Division: 2.0

QUESTION NO 04: Create a BankAccount Class

class BankAccount:

    def __init__(self, account_holder, balance):
        self.account_holder = account_holder
        self.balance = balance

    def deposit(self, amount):
        self.balance += amount
        print("Amount deposited:", amount)

    def withdraw(self, amount):
        if amount <= self.balance:
            self.balance -= amount
            print("Amount withdrawn:", amount)
        else:
            print("Insufficient balance!")

    def display_balance(self):
        print("Account Holder:", self.account_holder)
        print("Current Balance:", self.balance)


# Creating an object
account1 = BankAccount("Fiza", 5000)

# Calling methods
account1.display_balance()

account1.deposit(2000)

account1.withdraw(1000)

account1.display_balance()

OUTPUT:
Account Holder: Fiza
Current Balance: 5000

Amount deposited: 2000

Amount withdrawn: 1000

Account Holder: Fiza
Current Balance: 6000
